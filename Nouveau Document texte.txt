# =============================================================================
# src/ui/mise_en_page.py
# -----------------------------------------------------------------------------
# Rôle : Widget d'affichage et de mise en page du projet.
#
# ⚠️ CONVENTION D'ORIGINE ET D'AXES :
#   L'origine machine (0,0) est TOUJOURS au point SCÈNE (0, 0).
#   Le rectangle de zone s'étend DEPUIS l'origine dans le sens des axes :
#
#     Fond-Gauche  → X+ droite, Y+ bas
#     Fond-Droit   → X+ gauche, Y+ bas
#     Face-Gauche  → X+ droite, Y+ haut
#     Face-Droit   → X+ gauche, Y+ haut
#
#   "Fond" = côté opposé à l'opérateur. "Face" = côté opérateur.
#
# ⚠️ RÈGLES GRADUÉES (refonte — Étape 2F-1b) :
#   Les règles sont désormais des WIDGETS SÉPARÉS (voir src/ui/regles.py),
#   placés autour de la vue via un QGridLayout. Ça évite les artefacts
#   que drawForeground laissait en cas de repaint partiel (déplacement de
#   fenêtre, changement d'origine...).
#
# ⚠️ NOTION DE MASQUE :
#   Un masque = une couleur + paramètres P/V/N.
#   Import DXF → entités en noir sous le masque 'Import'.
#
# ⚠️ CHARGEMENT DXF (mode AJOUT) :
#   Charger un DXF AJOUTE au projet. Position absolue.
#
# ⚠️ DÉPLACEMENT D'ENTITÉS :
#   Flèches (boutons ou clavier) + drag souris SUR entité.
#   Modifie les COORDONNÉES RÉELLES (ent.points).
#
# ÉTAPES de la roadmap :
#   Étape 1..2E (FAIT) : affichage, masques, chargement, déplacement
#   Étape 2F-1b(FAIT) : règles graduées en widgets séparés (anti-artefacts)
#   Étape 2F-2 (TODO) : affichage coordonnées sélection + saisie cible
#   Étape 4     (TODO) : ordre de coupe
#   Étape 5     (TODO) : génération de G-code
# =============================================================================

class BoutonCouleur(QPushButton):
    """Bouton carré coloré pour le dialogue de création de masque."""

    def __init__(self, couleur_hex, parent=None):
        super().__init__(parent)
        self.couleur_hex = couleur_hex
        self.setFixedSize(40, 40)
        self.setCheckable(True)
        self._maj_style()

    def _maj_style(self):
        bordure = "3px solid #000000" if self.isChecked() else "1px solid #888888"
        self.setStyleSheet(
            f"QPushButton {{ background-color: {self.couleur_hex}; "
            f"border: {bordure}; border-radius: 4px; }}"
        )

    def setChecked(self, checked):
        super().setChecked(checked)
        self._maj_style()


class DialogueCreationMasque(QDialog):
    """Dialogue de création de masque : nom + choix parmi les 8 couleurs."""

    def __init__(self, nom_defaut="", parent=None):
        super().__init__(parent)
        self.setWindowTitle("Créer un masque")
        self.setMinimumWidth(400)

        self.couleur_choisie = None

        layout = QVBoxLayout(self)

        form = QFormLayout()
        self.champ_nom = QLineEdit(nom_defaut)
        form.addRow("Nom du masque :", self.champ_nom)
        layout.addLayout(form)

        label_couleur = QLabel("Couleur :")
        layout.addWidget(label_couleur)

        grille = QGrid()
        grille.setSpacing(6)
        self.boutons_couleur = []

        for i, (nom_couleur, hex_code) in enumerate(PALETTE_MASQUES):
            btn = BoutonCouleur(hex_code)
            btn.setToolTip(nom_couleur)
            btn.clicked.connect(lambda checked=False, idx=i: self._choisir_couleur(idx))
            grille.addWidget(btn, i // 4, i % 4)
            self.boutons_couleur.append(btn)

        layout.addLayout(grille)

        boutons = QDialogButtonBox(QDialogButtonBox.Ok | QDialogButtonBox.Cancel)
        boutons.accepted.connect(self.accept)
        boutons.rejected.connect(self.reject)
        layout.addWidget(boutons)

        self._choisir_couleur(0)

    def _choisir_couleur(self, index):
        for i, btn in enumerate(self.boutons_couleur):
            btn.setChecked(i == index)
        self.couleur_choisie = PALETTE_MASQUES[index][1]

    def get_resultat(self):
        nom = self.champ_nom.text().strip()
        if not nom:
            return None, None
        return nom, self.couleur_choisie


class EntiteItem(QGraphicsPathItem):
    """Item graphique d'une entité DXF, sélectionnable."""

    def __init__(self, entite, path, couleur):
        super().__init__(path)
        self.entite = entite
        self.couleur_base = couleur
        self.setFlag(QGraphicsPathItem.ItemIsSelectable, True)
        self.setAcceptHoverEvents(False)

        pen = QPen(couleur)
        pen.setWidthF(EPAISSEUR_NORMALE)
        pen.setCosmetic(True)
        self.setPen(pen)

        self.appliquer_style()

    def appliquer_style(self):
        if self.isSelected():
            pen = QPen(COULEUR_SELECTION)
            pen.setWidthF(EPAISSEUR_SELECTION)
            pen.setCosmetic(True)
            self.setPen(pen)
        else:
            pen = QPen(self.couleur_base)
            pen.setWidthF(EPAISSEUR_NORMALE)
            pen.setCosmetic(True)
            self.setPen(pen)

    def set_couleur(self, couleur):
        self.couleur_base = couleur
        self.appliquer_style()


class VueMiseEnPage(QGraphicsView):
    """
    QGraphicsView avec zoom, pan, sélection, drag.
    Émet le signal 'vue_changee' quand le zoom, le pan ou la taille change,
    pour permettre aux règles de se resynchroniser.
    """

    vue_changee = Signal()

    def __init__(self, scene):
        super().__init__(scene)
        self.setRenderHint(QPainter.Antialiasing)
        self.setTransformationAnchor(QGraphicsView.NoAnchor)
        self.setResizeAnchor(QGraphicsView.NoAnchor)
        self.setDragMode(QGraphicsView.NoDrag)
        self.setCursor(Qt.ArrowCursor)
        self.setFocusPolicy(Qt.StrongFocus)

        self._pan_actif = False
        self._pan_dernier_point = QPoint()
        self._espace_enfonce = False

        self._rect_select_actif = False
        self._rect_select_origine = QPointF()
        self._rect_select_item = None

        self._drag_actif = False
        self._drag_origine_scene = QPointF()
        self._drag_deja_bouge = False
        self._items_en_drag = []
        self._positions_origine = {}

        self.widget_parent = None

    # -------------------------------------------------------------------------
    # Zoom
    # -------------------------------------------------------------------------

    def wheelEvent(self, event):
        point_vue = event.position().toPoint()
        point_scene_avant = self.mapToScene(point_vue)
        facteur = FACTEUR_ZOOM if event.angleDelta().y() > 0 else 1 / FACTEUR_ZOOM
        self.scale(facteur, facteur)
        point_scene_apres = self.mapToScene(point_vue)
        delta = point_scene_apres - point_scene_avant
        centre_actuel = self.mapToScene(self.viewport().rect().center())
        self.centerOn(centre_actuel - delta)
        self.vue_changee.emit()
        event.accept()

    def resizeEvent(self, event):
        super().resizeEvent(event)
        self.vue_changee.emit()

    def scrollContentsBy(self, dx, dy):
        super().scrollContentsBy(dx, dy)
        self.vue_changee.emit()

    # -------------------------------------------------------------------------
    # Clavier
    # -------------------------------------------------------------------------

    def keyPressEvent(self, event):
        if event.key() == Qt.Key_Space and not event.isAutoRepeat():
            self._espace_enfonce = True
            self.setCursor(Qt.OpenHandCursor)
            event.accept()
            return

        if event.key() in (Qt.Key_Delete, Qt.Key_Backspace):
            if self.widget_parent is not None:
                self.widget_parent.supprimer_selection()
            event.accept()
            return

        if event.key() in (Qt.Key_Left, Qt.Key_Right, Qt.Key_Up, Qt.Key_Down):
            if self.widget_parent is not None:
                facteur = 10.0 if (event.modifiers() & Qt.ShiftModifier) else 1.0
                pas = self.widget_parent.pas_courant() * facteur
                sens = self.widget_parent.sens_axes()
                if event.key() == Qt.Key_Left:
                    self.widget_parent.deplacer_selection(-pas * sens['ex'], 0)
                elif event.key() == Qt.Key_Right:
                    self.widget_parent.deplacer_selection(pas * sens['ex'], 0)
                elif event.key() == Qt.Key_Up:
                    self.widget_parent.deplacer_selection(0, -pas * sens['ey'])
                elif event.key() == Qt.Key_Down:
                    self.widget_parent.deplacer_selection(0, pas * sens['ey'])
            event.accept()
            return

        super().keyPressEvent(event)

    def keyReleaseEvent(self, event):
        if event.key() == Qt.Key_Space and not event.isAutoRepeat():
            self._espace_enfonce = False
            if not self._pan_actif:
                self.setCursor(Qt.ArrowCursor)
            event.accept()
        else:
            super().keyReleaseEvent(event)

    # -------------------------------------------------------------------------
    # Souris
    # -------------------------------------------------------------------------

    def mousePressEvent(self, event):
        if event.button() == Qt.MiddleButton or \
           (event.button() == Qt.LeftButton and self._espace_enfonce):
            self._pan_actif = True
            self._pan_dernier_point = event.position().toPoint()
            self.setCursor(Qt.ClosedHandCursor)
            event.accept()
            return

        if event.button() == Qt.LeftButton:
            point_scene = self.mapToScene(event.position().toPoint())
            item = self.scene().itemAt(point_scene, self.transform())

            cible = item
            while cible is not None and not isinstance(cible, EntiteItem):
                cible = cible.parentItem()

            if cible is not None:
                modifiante = event.modifiers() & (Qt.ControlModifier | Qt.ShiftModifier)

                if modifiante:
                    cible.setSelected(not cible.isSelected())
                    cible.appliquer_style()
                    event.accept()
                    return

                if not cible.isSelected():
                    for it in self.scene().items():
                        if isinstance(it, EntiteItem) and it is not cible:
                            it.setSelected(False)
                            it.appliquer_style()
                    cible.setSelected(True)
                    cible.appliquer_style()

                self._drag_actif = True
                self._drag_deja_bouge = False
                self._drag_origine_scene = point_scene
                self._items_en_drag = [it for it in self.scene().items()
                                       if isinstance(it, EntiteItem) and it.isSelected()]
                self._positions_origine = {}
                for it in self._items_en_drag:
                    self._positions_origine[it] = it.pos()

                self.setCursor(Qt.SizeAllCursor)
                event.accept()
                return

            if not (event.modifiers() & (Qt.ControlModifier | Qt.ShiftModifier)):
                for it in self.scene().items():
                    if isinstance(it, EntiteItem):
                        it.setSelected(False)
                        it.appliquer_style()

            self._rect_select_actif = True
            self._rect_select_origine = point_scene

            self._rect_select_item = QGraphicsRectItem()
            pen = QPen(COULEUR_BORD_RECT_SEL)
            pen.setWidthF(0)
            pen.setCosmetic(True)
            pen.setStyle(Qt.DashLine)
            self._rect_select_item.setPen(pen)
            self._rect_select_item.setBrush(QBrush(COULEUR_RECT_SELECT))
            self.scene().addItem(self._rect_select_item)

            event.accept()
            return

        super().mousePressEvent(event)

    def mouseMoveEvent(self, event):
        if self._pan_actif:
            point_actuel = event.position().toPoint()
            delta = point_actuel - self._pan_dernier_point
            self._pan_dernier_point = point_actuel
            self.horizontalScrollBar().setValue(
                self.horizontalScrollBar().value() - delta.x()
            )
            self.verticalScrollBar().setValue(
                self.verticalScrollBar().value() - delta.y()
            )
            event.accept()
            return

        if self._drag_actif:
            point_scene_actuel = self.mapToScene(event.position().toPoint())
            delta = point_scene_actuel - self._drag_origine_scene

            if not self._drag_deja_bouge:
                dist_ecran = (event.position().toPoint() -
                              self.mapFromScene(self._drag_origine_scene)).manhattanLength()
                if dist_ecran < SEUIL_DU_DRAG:
                    event.accept()
                    return
                self._drag_deja_bouge = True

            for it in self._items_en_drag:
                pos_init = self._positions_origine[it]
                it.setPos(pos_init + delta)

            event.accept()
            return

        if self._rect_select_actif:
            point_scene_actuel = self.mapToScene(event.position().toPoint())
            rect = QRectF(self._rect_select_origine, point_scene_actuel).normalized()
            self._rect_select_item.setRect(rect)
            event.accept()
            return

        super().mouseMoveEvent(event)

    def mouseReleaseEvent(self, event):
        if self._pan_actif and (event.button() == Qt.MiddleButton or
                                event.button() == Qt.LeftButton):
            self._pan_actif = False
            if self._espace_enfonce:
                self.setCursor(Qt.OpenHandCursor)
            else:
                self.setCursor(Qt.ArrowCursor)
            event.accept()
            return

        if self._drag_actif and event.button() == Qt.LeftButton:
            self._drag_actif = False
            self.setCursor(Qt.ArrowCursor)

            if self._drag_deja_bouge:
                point_scene_actuel = self.mapToScene(event.position().toPoint())
                delta = point_scene_actuel - self._drag_origine_scene
                dx_scene = delta.x()
                dy_scene = delta.y()

                if self.widget_parent is not None:
                    self.widget_parent.deplacer_selection_par_delta_scene(
                        dx_scene, dy_scene, self._items_en_drag
                    )

            self._items_en_drag = []
            self._positions_origine = {}
            self._drag_deja_bouge = False
            event.accept()
            return

        if self._rect_select_actif and event.button() == Qt.LeftButton:
            point_scene_fin = self.mapToScene(event.position().toPoint())
            rect = QRectF(self._rect_select_origine, point_scene_fin).normalized()

            point_debut_ecran = self.mapFromScene(self._rect_select_origine)
            point_fin_ecran = self.mapFromScene(point_scene_fin)
            dist = (point_fin_ecran - point_debut_ecran).manhattanLength()

            if self._rect_select_item is not None:
                self.scene().removeItem(self._rect_select_item)
                self._rect_select_item = None

            if dist > SEUIL_DU_PAN:
                for it in self.scene().items():
                    if isinstance(it, EntiteItem):
                        if rect.contains(it.boundingRect()):
                            it.setSelected(True)
                            it.appliquer_style()

            self._rect_select_actif = False
            event.accept()
            return

        super().mouseReleaseEvent(event)

    # -------------------------------------------------------------------------
    # Synchronisation des règles
    # -------------------------------------------------------------------------

    def _maj_regles(self):
        """
        Met à jour les deux widgets règles avec l'état actuel de la vue.
        """
        vue = self.view
        vp = vue.viewport()

        if vp.width() <= 0 or vp.height() <= 0:
            return

        # Pixels par mm à l'échelle actuelle
        pixels_par_mm = vue.transform().m11()
        if pixels_par_mm <= 0:
            return

        # Coordonnées scène des bords visibles du viewport
        pt_hg = vue.mapToScene(0, 0)
        pt_bd = vue.mapToScene(vp.width(), vp.height())

        scene_x_min = pt_hg.x()
        scene_x_max = pt_bd.x()
        scene_y_min = pt_hg.y()
        scene_y_max = pt_bd.y()

        sens = self.sens_axes()

        # Règle horizontale
        self.regle_h.maj_etat(
            pixels_par_mm,
            scene_x_min, scene_x_max,
            0,           # offset_px (les deux widgets ont la même origine)
            sens['ex']
        )

        # Règle verticale
        self.regle_v.maj_etat(
            pixels_par_mm,
            scene_y_min, scene_y_max,
            0,
            sens['ey']
        )

    # -------------------------------------------------------------------------
    # Sens des axes
    # -------------------------------------------------------------------------

    def sens_axes(self):
        if self.origine_zone == "fond_gauche":
            return {'ex': 1, 'ey': 1}
        elif self.origine_zone == "fond_droit":
            return {'ex': -1, 'ey': 1}
        elif self.origine_zone == "face_gauche":
            return {'ex': 1, 'ey': -1}
        else:
            return {'ex': -1, 'ey': -1}

    def origine_vers_scene(self, x_machine, y_machine):
        sens = self.sens_axes()
        return QPointF(x_machine * sens['ex'], y_machine * sens['ey'])

    # -------------------------------------------------------------------------
    # Zone de travail et repère
    # -------------------------------------------------------------------------

    def appliquer_zone(self):
        self.largeur_zone = self.spin_largeur.value()
        self.hauteur_zone = self.spin_hauteur.value()
        self.mettre_a_jour_zone_travail()
        self._rafraichir_items_depuis_modeles()
        self.recadrer_sur_projet()
        self._maj_regles()

    def changer_origine(self):
        self.origine_zone = self.combo_origine.currentData()
        self.mettre_a_jour_zone_travail()
        self._rafraichir_items_depuis_modeles()
        self._maj_regles()

    def mettre_a_jour_zone_travail(self):
        sens = self.sens_axes()
        L = self.largeur_zone * sens['ex']
        H = self.hauteur_zone * sens['ey']

        x1 = min(0, L)
        y1 = min(0, H)
        x2 = max(0, L)
        y2 = max(0, H)
        self.item_zone_travail.setRect(x1, y1, x2 - x1, y2 - y1)

        ox, oy = 0.0, 0.0
        sx, sy = sens['ex'], sens['ey']

        t = TAILLE_FLECHE
        long = LONGUEUR_AXE

        fin_x = ox + long * sx
        path_x = QPainterPath()
        path_x.moveTo(ox, oy)
        path_x.lineTo(fin_x, oy)
        self.item_fleche_x.setPath(path_x)

        pointe_x = QPolygonF([
            QPointF(fin_x, oy),
            QPointF(fin_x - t * sx, oy - t / 2),
            QPointF(fin_x - t * sx, oy + t / 2),
        ])
        self.item_pointe_x.setPolygon(pointe_x)
        self.item_texte_x.setPos(fin_x + 2 * sx, oy - 8)

        fin_y = oy + long * sy
        path_y = QPainterPath()
        path_y.moveTo(ox, oy)
        path_y.lineTo(ox, fin_y)
        self.item_fleche_y.setPath(path_y)

        pointe_y = QPolygonF([
            QPointF(ox, fin_y),
            QPointF(ox - t / 2, fin_y - t * sy),
            QPointF(ox + t / 2, fin_y - t * sy),
        ])
        self.item_pointe_y.setPolygon(pointe_y)
        self.item_texte_y.setPos(ox + 4, fin_y + 2 * sy)

    def _rafraichir_items_depuis_modeles(self):
        for it in self.items_entites:
            ent = it.entite
            path = QPainterPath()
            if not ent.points:
                continue
            pts_scene = [self.origine_vers_scene(p[0], p[1]) for p in ent.points]
            premier = pts_scene[0]
            path.moveTo(premier.x(), premier.y())
            for p in pts_scene[1:]:
                path.lineTo(p.x(), p.y())
            it.setPath(path)
            it.setPos(QPointF(0, 0))

    # -------------------------------------------------------------------------
    # Déplacement des entités
    # -------------------------------------------------------------------------

    def pas_courant(self):
        return self.combo_pas.currentData()

    def _deplacer_via_bouton(self, signe_x_ecran, signe_y_ecran):
        pas = self.pas_courant()
        sens = self.sens_axes()
        dx_machine = signe_x_ecran * pas * sens['ex']
        dy_machine = signe_y_ecran * pas * sens['ey']
        self.deplacer_selection(dx_machine, dy_machine)

    def deplacer_selection(self, dx_machine, dy_machine):
        selection = [it for it in self.items_entites if it.isSelected()]
        if not selection:
            return

        for it in selection:
            ent = it.entite
            ent.points = [(p[0] + dx_machine, p[1] + dy_machine)
                          for p in ent.points]

        self._rafraichir_items_depuis_modeles()
        self._mettre_a_jour_zone_cadrage()

    def deplacer_selection_par_delta_scene(self, dx_scene, dy_scene, items):
        if not items:
            return

        sens = self.sens_axes()
        dx_machine = dx_scene * sens['ex']
        dy_machine = dy_scene * sens['ey']

        for it in items:
            ent = it.entite
            ent.points = [(p[0] + dx_machine, p[1] + dy_machine)
                          for p in ent.points]

        self._rafraichir_items_depuis_modeles()
        self._mettre_a_jour_zone_cadrage()

    # -------------------------------------------------------------------------
    # Gestion des masques
    # -------------------------------------------------------------------------

    def _masque_existe(self, nom):
        return any(m.nom == nom for m in self.masques)

    def _trouver_masque(self, nom):
        for m in self.masques:
            if m.nom == nom:
                return m
        return None

    def _compte_entites_par_masque(self):
        comptes = {}
        for it in self.items_entites:
            nom = it.entite.calque
            comptes[nom] = comptes.get(nom, 0) + 1
        return comptes

    def _rafraichir_liste_masques(self):
        nom_selectionne = None
        item_courant = self.liste_masques.currentItem()
        if item_courant is not None:
            nom_selectionne = item_courant.data(Qt.UserRole)

        self.liste_masques.clear()
        comptes = self._compte_entites_par_masque()

        for masque in self.masques:
            n = comptes.get(masque.nom, 0)
            texte = f"{masque.nom}   ({n})"

            item = QListWidgetItem(texte)
            item.setData(Qt.UserRole, masque.nom)

            pixmap = QPixmap(12, 12)
            pixmap.fill(masque.couleur)
            item.setIcon(QIcon(pixmap))

            self.liste_masques.addItem(item)

            if masque.nom == nom_selectionne:
                self.liste_masques.setCurrentItem(item)

    def creer_masque(self):
        index_nouveau = len(self.masques)
        nom_defaut = f"Masque {index_nouveau + 1}"

        dialogue = DialogueCreationMasque(nom_defaut, self)
        if dialogue.exec() != QDialog.Accepted:
            return

        nom, couleur_hex = dialogue.get_resultat()
        if not nom:
            return

        if self._masque_existe(nom):
            QMessageBox.warning(self, "Doublon",
                                f"Un masque nommé « {nom} » existe déjà.")
            return

        self.masques.append(Masque(nom, couleur_hex))
        self._rafraichir_liste_masques()
        self._maj_label_info()

    def _on_masque_selectionne(self, row):
        if row < 0:
            self.spin_puissance.setValue(0)
            self.spin_vitesse.setValue(0)
            self.spin_passes.setValue(1)
            return

        item = self.liste_masques.item(row)
        if item is None:
            return
        nom = item.data(Qt.UserRole)
        masque = self._trouver_masque(nom)
        if masque is None:
            return

        self.spin_puissance.blockSignals(True)
        self.spin_vitesse.blockSignals(True)
        self.spin_passes.blockSignals(True)

        self.spin_puissance.setValue(masque.puissance)
        self.spin_vitesse.setValue(masque.vitesse)
        self.spin_passes.setValue(masque.passes)

        self.spin_puissance.blockSignals(False)
        self.spin_vitesse.blockSignals(False)
        self.spin_passes.blockSignals(False)

    def _on_parametres_modifies(self):
        item = self.liste_masques.currentItem()
        if item is None:
            return
        nom = item.data(Qt.UserRole)
        masque = self._trouver_masque(nom)
        if masque is None:
            return

        masque.puissance = self.spin_puissance.value()
        masque.vitesse = self.spin_vitesse.value()
        masque.passes = self.spin_passes.value()

    def assigner_au_masque(self):
        item = self.liste_masques.currentItem()
        if item is None:
            QMessageBox.information(self, "Aucun masque",
                                    "Sélectionnez un masque dans la liste.")
            return

        nom_masque = item.data(Qt.UserRole)
        masque = self._trouver_masque(nom_masque)
        if masque is None:
            return

        selection = [it for it in self.items_entites if it.isSelected()]
        if not selection:
            QMessageBox.information(self, "Aucune sélection",
                                    "Sélectionnez au moins une entité.")
            return

        for it in selection:
            it.entite.calque = nom_masque
            it.set_couleur(masque.couleur)

        self._rafraichir_liste_masques()
        self._maj_label_info()

    def renommer_masque(self):
        item = self.liste_masques.currentItem()
        if item is None:
            return

        ancien_nom = item.data(Qt.UserRole)
        masque = self._trouver_masque(ancien_nom)
        if masque is None:
            return

        nouveau_nom, ok = QInputDialog.getText(
            self, "Renommer le masque", "Nouveau nom :",
            text=ancien_nom
        )
        if not ok or not nouveau_nom.strip():
            return

        nouveau_nom = nouveau_nom.strip()

        if nouveau_nom == ancien_nom:
            return
        if self._masque_existe(nouveau_nom):
            QMessageBox.warning(self, "Doublon",
                                f"Un masque nommé « {nouveau_nom} » existe déjà.")
            return

        masque.nom = nouveau_nom
        for it in self.items_entites:
            if it.entite.calque == ancien_nom:
                it.entite.calque = nouveau_nom

        self._rafraichir_liste_masques()
        self._maj_label_info()

    def supprimer_masque(self):
        item = self.liste_masques.currentItem()
        if item is None:
            return

        nom = item.data(Qt.UserRole)
        masque = self._trouver_masque(nom)
        if masque is None:
            return

        n_entites = sum(1 for it in self.items_entites
                        if it.entite.calque == nom)

        if n_entites > 0:
            message = (f"Supprimer le masque « {nom} » ?\n\n"
                       f"{n_entites} entité(s) seront réassignées au masque "
                       f"« {NOM_MASQUE_IMPORT} ».")
        else:
            message = f"Supprimer le masque « {nom} » ?"

        reponse = QMessageBox.question(
            self, "Confirmer la suppression",
            message,
            QMessageBox.Yes | QMessageBox.No,
            QMessageBox.No
        )
        if reponse != QMessageBox.Yes:
            return

        if n_entites > 0 and not self._masque_existe(NOM_MASQUE_IMPORT):
            self.masques.append(Masque(NOM_MASQUE_IMPORT, COULEUR_IMPORT))

        masque_import = self._trouver_masque(NOM_MASQUE_IMPORT)

        for it in self.items_entites:
            if it.entite.calque == nom:
                it.entite.calque = NOM_MASQUE_IMPORT
                if masque_import is not None:
                    it.set_couleur(masque_import.couleur)

        self.masques = [m for m in self.masques if m.nom != nom]

        self._rafraichir_liste_masques()
        self._maj_label_info()

    # -------------------------------------------------------------------------
    # Chargement DXF (mode AJOUT)
    # -------------------------------------------------------------------------

    def charger_dxf(self):
        chemin, _ = QFileDialog.getOpenFileName(
            self, "Choisir un fichier DXF", "",
            "Fichiers DXF (*.dxf);;Tous les fichiers (*.*)"
        )
        if not chemin:
            return

        try:
            from src.core.dxf_loader import charger_dxf
        except ImportError as e:
            QMessageBox.critical(self, "Erreur",
                                 f"Impossible d'importer le lecteur DXF :\n{e}")
            return

        try:
            entites, warnings = charger_dxf(chemin)
        except Exception as e:
            QMessageBox.critical(self, "Erreur de lecture DXF",
                                 f"Impossible de lire le fichier :\n{e}")
            return

        self.ajouter_entites(entites, warnings)

        nom = os.path.basename(chemin)
        self.label_info.setText(
            f"{nom} ajouté  —  {len(self.items_entites)} entité(s) au total"
        )

    def ajouter_entites(self, entites, warnings=None):
        if not self._masque_existe(NOM_MASQUE_IMPORT):
            self.masques.append(Masque(NOM_MASQUE_IMPORT, COULEUR_IMPORT))

        for ent in entites:
            ent.calque = NOM_MASQUE_IMPORT

        couleurs_masques = {m.nom: m.couleur for m in self.masques}

        zone_totale = QRectF()

        for ent in entites:
            path = QPainterPath()
            if not ent.points:
                continue
            pts_scene = [self.origine_vers_scene(p[0], p[1]) for p in ent.points]
            premier = pts_scene[0]
            path.moveTo(premier.x(), premier.y())
            for p in pts_scene[1:]:
                path.lineTo(p.x(), p.y())

            couleur = couleurs_masques.get(ent.calque, QColor("#000000"))
            item = EntiteItem(ent, path, couleur)
            self.scene.addItem(item)
            self.items_entites.append(item)

            r = path.boundingRect()
            zone_totale = zone_totale.united(r) if not zone_totale.isNull() else r

        self._mettre_a_jour_zone_cadrage()
        QTimer.singleShot(0, self.recadrer_sur_projet)
        QTimer.singleShot(50, self._maj_regles)

        self._rafraichir_liste_masques()
        self._maj_label_info()

        if warnings:
            QMessageBox.information(
                self, "Chargement terminé avec avertissements",
                "\n".join(warnings)
            )

    # -------------------------------------------------------------------------

    def vider_projet(self):
        reponse = QMessageBox.question(
            self,
            "Vider le projet",
            "Vider tout le projet ?\n\n"
            "Les entités, les masques, la zone de travail et l'origine "
            "seront remis à leurs valeurs par défaut.\n\n"
            "Cette action ne peut pas être annulée.",
            QMessageBox.Yes | QMessageBox.No,
            QMessageBox.No
        )
        if reponse != QMessageBox.Yes:
            return

        for item in self.items_entites:
            self.scene.removeItem(item)
        self.items_entites = []
        self.masques = []

        self.largeur_zone = ZONE_DEFAUT_LARGEUR
        self.hauteur_zone = ZONE_DEFAUT_HAUTEUR
        self.spin_largeur.blockSignals(True)
        self.spin_hauteur.blockSignals(True)
        self.spin_largeur.setValue(ZONE_DEFAUT_LARGEUR)
        self.spin_hauteur.setValue(ZONE_DEFAUT_HAUTEUR)
        self.spin_largeur.blockSignals(False)
        self.spin_hauteur.blockSignals(False)

        self.origine_zone = ORIGINE_DEFAUT
        self.combo_origine.blockSignals(True)
        for idx, (_, cle) in enumerate(ORIGINES_DISPONIBLES):
            if cle == ORIGINE_DEFAUT:
                self.combo_origine.setCurrentIndex(idx)
                break
        self.combo_origine.blockSignals(False)

        self.mettre_a_jour_zone_travail()
        self._rafraichir_liste_masques()
        self.label_info.setText("Projet vide")
        self._maj_label_info()

        self.zone_projet = QRectF()
        self.recadrer_sur_projet()
        QTimer.singleShot(50, self._maj_regles)

    # -------------------------------------------------------------------------
    # Cadrage
    # -------------------------------------------------------------------------

    def _mettre_a_jour_zone_cadrage(self, zone_totale=None):
        sens = self.sens_axes()
        L = self.largeur_zone * sens['ex']
        H = self.hauteur_zone * sens['ey']
        zone_zone_travail = QRectF(
            min(0, L), min(0, H), abs(L), abs(H)
        )

        if zone_totale is None:
            zone_totale = QRectF()
            for it in self.items_entites:
                r = it.boundingRect()
                zone_totale = zone_totale.united(r) if not zone_totale.isNull() else r

        if zone_totale.isNull() or zone_totale.isEmpty():
            zone_finale = zone_zone_travail
        else:
            zone_finale = zone_totale.united(zone_zone_travail)

        marge_x = max(zone_finale.width(), 1.0) * MARGE_VUE
        marge_y = max(zone_finale.height(), 1.0) * MARGE_VUE
        zone_finale.adjust(-marge_x, -marge_y, marge_x, marge_y)
        self.zone_projet = zone_finale

        centre = self.zone_projet.center()
        largeur = max(self.zone_projet.width(), 1.0) * FACTEUR_ELARGISSEMENT_SCENE
        hauteur = max(self.zone_projet.height(), 1.0) * FACTEUR_ELARGISSEMENT_SCENE
        self.scene.setSceneRect(QRectF(
            centre.x() - largeur / 2,
            centre.y() - hauteur / 2,
            largeur, hauteur
        ))
        
    def recadrer_sur_projet(self):
        sens = self.sens_axes()
        L = self.largeur_zone * sens['ex']
        H = self.hauteur_zone * sens['ey']
        zone_zone_travail = QRectF(
            min(0, L), min(0, H), abs(L), abs(H)
        )

        if self.zone_projet.isNull() or self.zone_projet.isEmpty():
            cible = zone_zone_travail
        else:
            cible = self.zone_projet

        marge_x = max(cible.width(), 1.0) * 0.05
        marge_y = max(cible.height(), 1.0) * 0.05
        cible = QRectF(cible)
        cible.adjust(-marge_x, -marge_y, marge_x, marge_y)

        # Astuce anti-timing : forcer Qt à recalculer la taille du viewport
        self.view.viewport().update()
        self.scene.update()


        self.view.fitInView(cible, Qt.KeepAspectRatio)
        QTimer.singleShot(50, self._maj_regles)

    # -------------------------------------------------------------------------
    # Suppression d'entités
    # -------------------------------------------------------------------------

    def supprimer_selection(self):
        selection = [it for it in self.items_entites if it.isSelected()]
        n = len(selection)

        if n == 0:
            return

        reponse = QMessageBox.question(
            self,
            "Confirmer la suppression",
            f"Supprimer {n} entité(s) sélectionnée(s) ?\n\n"
            f"Cette action ne peut pas être annulée.",
            QMessageBox.Yes | QMessageBox.No,
            QMessageBox.No
        )
        if reponse != QMessageBox.Yes:
            return

        for it in selection:
            self.scene.removeItem(it)
            if it in self.items_entites:
                self.items_entites.remove(it)

        masques_utilises = set(it.entite.calque for it in self.items_entites)
        self.masques = [m for m in self.masques if m.nom in masques_utilises]

        self._mettre_a_jour_zone_cadrage()
        self._rafraichir_liste_masques()
        self._maj_label_info()

    def _maj_label_info(self):
        n = len(self.items_entites)
        n_masques = len(self.masques)
        if n == 0:
            self.label_info.setText(f"Aucune entité  —  {n_masques} masque(s)")
        else:
            self.label_info.setText(
                f"{n} entité(s)  —  {n_masques} masque(s)"
            )

    def _maj_bouton_supprimer(self):
        selection = [it for it in self.items_entites if it.isSelected()]
        self.btn_supprimer.setEnabled(len(selection) > 0)
